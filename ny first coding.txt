campaigns = [
    ["Facebook", 500, 10000, 100, 50],
    ["Instagram", 700, 15000, 150, 80],
    ["Google Ads", 1000, 20000, 200, 120]
]

print("AI MARKETING ANALYTICS PLATFORM")
print()

best_roi = 0
best_campaign = ""

for campaign in campaigns:
    name = campaign[0]
    clicks = campaign[1]
    impressions = campaign[2]
    cost = campaign[3]
    sales = campaign[4]

    ctr = (clicks / impressions) * 100
    roi = sales / cost

    print("Campaign:", name)
    print("CTR:", round(ctr, 2), "%")
    print("ROI:", round(roi, 2))
    print()

    if roi > best_roi:
        best_roi = roi
        best_campaign = name

print("Best Campaign:", best_campaign)

print("\nAI Recommendation:")
print("Increase budget for", best_campaign)
print("Target age group 25-35")
print("Run ads during evening hours")
